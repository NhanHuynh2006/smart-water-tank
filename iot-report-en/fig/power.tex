% Power distribution and pump circuit, drawn in TikZ with straight, right-angled wires.
\begingroup\setstretch{1.0}
\begin{tikzpicture}[
    font=\small,
    box/.style={draw, thick, rounded corners=2pt, align=center, fill=white,
                minimum width=3.1cm, minimum height=1.05cm},
    term/.style={draw, thick, fill=black!6, align=center, minimum height=0.6cm},
    arr/.style={-{Stealth[length=2.4mm]}, thick},
    wire/.style={thick},
    lbl/.style={font=\footnotesize, fill=white, inner sep=1pt, align=center}
]
% ---- supply chain ----
\node[box] (sup)  at (0,0)    {\textbf{12 V supply}};
\node[box] (fuse) at (4.2,0)  {\textbf{2 A fuse}};
\node[box] (buck) at (8.4,0)  {\textbf{Buck converter}\\5 V, 5 A};
\draw[arr] (sup) -- (fuse);
\draw[arr] (fuse) -- (buck);

% ---- 5 V output terminal: one screw per branch (star) ----
\node[term, minimum width=15.2cm] (vt) at (6.3,-1.9) {5 V output terminal of the buck converter: every branch leaves on its own wire};
\draw[arr] (buck) -- (buck |- vt.north);

% ---- loads ----
\node[box] (esp)  at (0,-3.9)    {\textbf{ESP32 DevKit}\\470 $\mu$F on the board};
\node[box] (sens) at (4.2,-3.9)  {\textbf{Sensor supply}\\HC-SR04, 2 $\times$ YF-S401,\\ACS712};
\node[box] (coil) at (8.4,-3.9)  {\textbf{Relay coil}\\driven from GPIO 10};
\node[box] (cont) at (12.6,-3.9) {\textbf{Relay contact}\\COM to NO};
\node[box] (acs)  at (12.6,-5.9) {\textbf{ACS712}\\IP+ to IP$-$};
\node[box] (pump) at (12.6,-7.9) {\textbf{5 V pump}\\1N4007 and 100 nF\\across the terminals};
\foreach \n in {esp, sens, coil, cont}
  \draw[arr] (\n |- vt.south) -- (\n.north);
\draw[arr] (cont) -- node[lbl, right=3pt] {switched 5 V} (acs);
\draw[arr] (acs) -- node[lbl, right=3pt] {measured\\pump current} (pump);

% ---- ground terminal ----
\node[term, minimum width=15.2cm] (gt) at (6.3,-9.8) {Ground terminal at the buck converter: the pump return does not share a wire with the controller};
\foreach \n in {esp, sens, coil, pump}
  \draw[wire] (\n.south) -- (\n |- gt.north);
\end{tikzpicture}
\endgroup
